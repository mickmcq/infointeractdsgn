\definecolor{ambitioncolor}{rgb}{0.90,0.10,0.10}

\tikzset{
  ambitionbubble/.style={
    align=center,
    draw=ambitioncolor,
    fill=ambitioncolor,
    thick,
%                                 center, not base, so the
%                                 bubble sits exactly on its
%                                 coordinate
    anchor=center,
    text=white,
    shape=ellipse,
    minimum size=3pt,
    opacity=0.95,
    inner sep=4pt,
    font=\ttfamily\small
  },
  ambitionarrow/.style={
    ->,
    >=stealth,
    very thick,
    opacity=0.9,
    draw=ambitioncolor,
%                                 the dvi route colors arrow
%                                 tips by fill, the pdf route
%                                 by draw, so set both
    fill=ambitioncolor
  },
  ambitioncaption/.style={
    align=center,
    anchor=base,
    text=chenattrcolor,
    font=\ttfamily\small
  }
}
