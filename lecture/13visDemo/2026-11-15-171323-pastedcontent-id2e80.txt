⏺ # Install pacman only if it's missing, then load packages
  if (!requireNamespace("pacman", quietly = TRUE)) install.packages("pacman")
  pacman::p_load(tidyverse, readxl)

  # Read the Orders tab
  orders <- read_excel("GlobalSuperstore.xls", sheet = "Orders")
  View(orders)

  # US furniture orders only
  us_furniture <- orders |>
    filter(Country == "United States", Category == "Furniture")

  # Six most frequently occurring states
  top_states <- us_furniture |>
    count(State, sort = TRUE) |>
    slice_head(n = 6) |>
    pull(State)

  # One row per customer per state: average profit across that customer's orders
  customer_profit <- us_furniture |>
    filter(State %in% top_states) |>
    group_by(State, `Customer ID`) |>
    summarize(Profit = mean(Profit), .groups = "drop")

  # Boxplots overlaid with jitter plots
  ggplot(customer_profit, aes(x = State, y = Profit)) +
    geom_boxplot(outlier.shape = NA) +
    geom_jitter(width = 0.2, alpha = 0.1) +
    labs(
      title = "Average furniture profit per customer, top six US states",
      x = "State",
      y = "Average profit per customer"
    )

  A few choices I made where the request was open:

  - Top six states: these are counted among US furniture orders, not all US
    orders. To rank by all US orders instead, move the Category == "Furniture"
    filter to after top_states is computed.
  - Customers in more than one state: averaging is per customer within each
    state, so such a customer gets one dot in each state.
  - Outliers: outlier.shape = NA hides the boxplot's own outlier points, since
    the jitter layer already draws every customer; without it, outliers would
    appear twice.
  - Alpha: alpha = 0.1 makes the dots very faint, which helps show where they
    overlap but can make sparse states hard to see. Raise it (for example to
    0.3) if the dots are too hard to see.
